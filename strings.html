21. Print all the permutation of a string i.e. "ABCDE".

Code:

#include <stdio.h>
#include <string.h>

void swap(char *a, char *b)
{
    char t = *a;
    *a = *b;
    *b = t;
}

void permute(char *s, int l, int r)
{
    int i;
    if (l == r)
    {
        printf("%s\n", s);
        return;
    }
    for (i = l; i <= r; i++)
    {
        swap(&s[l], &s[i]);
        permute(s, l + 1, r);
        swap(&s[l], &s[i]);
    }
}

int main()
{
    char s[] = "ABCDE";
    permute(s, 0, (int)strlen(s) - 1);
    return 0;
}

    /* Check duplicate word */
    cur = head;
    while (cur != NULL)
    {
        if (strcmp(cur->word, newnode->word) == 0)
        {
            printf("\nWord already exists.\n");
            free(newnode);
            return;
        }
        cur = cur->next;
    }
    /* First node */
    if (head == NULL)
    {
        head = newnode;
    }
    else
    {
        cur = head;
        while (cur->next != NULL)
        {
            cur = cur->next;
        }
        cur->next = newnode;
    }
    printf("\nWord added successfully.\n");
}

void deleteWord()
{
    char word[50];
    struct node *cur = head;
    struct node *prev = NULL;
    printf("\nEnter word to remove: ");
    scanf(" %49[^\n]", word);
    if (head == NULL)
    {
        printf("\nDictionary is empty.\n");
        return;
    }
    while (cur != NULL)
    {
        if (strcmp(cur->word, word) == 0)
        {
            if (cur == head)
            {
                head = head->next;
            }
            else
            {
                prev->next = cur->next;
            }
            free(cur);
            printf("\nWord removed successfully.\n");
            return;

        prev = cur;
        cur = cur->next;
    }
    printf("\nWord not found.\n");
}
