1. Enter a matrix of 4 rows and 2 columns from the user display it in matrix format?

Code:

#include <stdio.h>

int main()
{
    int m[4][2], i, j;
    printf("Enter elements of 4 x 2 matrix:\n");
    for (i = 0; i < 4; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &m[i][j]);
    printf("Matrix:\n");
    for (i = 0; i < 4; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%5d", m[i][j]);
        printf("\n");
    }
    return 0;
}

4. Enter two Matrices from user, multiply the matrices and display the answer?

Code:

#include <stdio.h>

int main()
{
    int a[10][10], b[10][10], p[10][10];
    int r1, c1, r2, c2, i, j, k;
    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1, &c1);
    printf("Enter rows and columns of second matrix: ");
    scanf("%d %d", &r2, &c2);
    if (c1 != r2)
    {
        printf("Multiplication not possible\n");
        return 0;
    }
    printf("Enter first matrix:\n");
    for (i = 0; i < r1; i++)
        for (j = 0; j < c1; j++)
            scanf("%d", &a[i][j]);
    printf("Enter second matrix:\n");
    for (i = 0; i < r2; i++)
        for (j = 0; j < c2; j++)
            scanf("%d", &b[i][j]);
    for (i = 0; i < r1; i++)
        for (j = 0; j < c2; j++)
        {
            p[i][j] = 0;
            for (k = 0; k < c1; k++)
                p[i][j] = p[i][j] + a[i][k] * b[k][j];
        }
    printf("Product:\n");
    for (i = 0; i < r1; i++)
    {
        for (j = 0; j < c2; j++)
            printf("%5d", p[i][j]);
        printf("\n");
    }
    return 0;
}

7. Create a tic tac toe game using 2D Arrays.

Code:

#include <stdio.h>

void showBoard(char b[3][3])
{
    int i;
    printf("\n");
    for (i = 0; i < 3; i++)
    {
        printf(" %c | %c | %c \n", b[i][0], b[i][1], b[i][2]);
        if (i < 2)
            printf("---+---+---\n");
    }
    printf("\n");
}

int isWinner(char b[3][3], char p)
{
    int i;
    for (i = 0; i < 3; i++)
    {
        if (b[i][0] == p && b[i][1] == p && b[i][2] == p)
            return 1;
        if (b[0][i] == p && b[1][i] == p && b[2][i] == p)
            return 1;
    }
    if (b[0][0] == p && b[1][1] == p && b[2][2] == p)
        return 1;
    if (b[0][2] == p && b[1][1] == p && b[2][0] == p)
        return 1;
    return 0;
}

int main()
{
    char b[3][3];
    char player = 'X';
    int i, j, pos, row, col, moves = 0;
    for (i = 0; i < 3; i++)
        for (j = 0; j < 3; j++)
            b[i][j] = '1' + (i * 3 + j);
    while (1)
    {
        showBoard(b);
        printf("Player %c, enter position (1-9): ", player);
        if (scanf("%d", &pos) != 1)
        {
            while (getchar() != '\n');
            printf("Invalid input\n");
            continue;
        }
        if (pos < 1 || pos > 9)
        {
            printf("Position must be 1 to 9\n");
            continue;
        }
        row = (pos - 1) / 3;
        col = (pos - 1) % 3;
        if (b[row][col] == 'X' || b[row][col] == 'O')
        {
            printf("Cell already taken\n");
            continue;
        }
        b[row][col] = player;
        moves++;
        if (isWinner(b, player))
        {
            showBoard(b);
            printf("Player %c wins\n", player);
            break;
        }
        if (moves == 9)
        {
            showBoard(b);
            printf("Match draw\n");
            break;
        }
        if (player == 'X')
            player = 'O';
        else
            player = 'X';
    }
    return 0;
}
