1. Enter a integer array from the user and perform selection sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Code:

#include <stdio.h>
#include <stdlib.h>

void printArray(int a[], int n)
{
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
}

int main(void)
{
    int n;
    scanf("%d", &n);
    int *a = malloc(n * sizeof(int));
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    for (int i = 0; i < n - 1; i++)
    {
        int min = i;
        for (int j = i + 1; j < n; j++)
            if (a[j] < a[min])
                min = j;
        int t = a[i];
        a[i] = a[min];
        a[min] = t;
    }
    printArray(a, n);
    printf("Best time complexity: O(n^2)\nWorst time complexity: O(n^2)\n");
    free(a);
    return 0;
}

7. Enter a integer array from the user and perform merge sort algorithm through iteration. also calculate the worst and best complexity with exact time.

#include <stdio.h>
#define MAX 1000

typedef int Item;

int cmp(Item a, Item b)
{
    return (a > b) -(a < b);
}

void merge(Item a[], Item tmp[], int l, int m, int r)
{
    int i = l, j = m + 1, k = l;
    while (i <= m && j <= r)
    {
        if (cmp(a[i], a[j]) <= 0)
            tmp[k++] = a[i++];
        else
            tmp[k++] = a[j++];
    }
    while (i <= m)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];
    for (int x = l; x <= r; x++)
        a[x] = tmp[x];
}

void mergeSort(Item a[], int n)
{
    Item tmp[MAX];
    for (int width = 1; width < n; width *= 2)
    {
        for (int left = 0; left < n; left += 2*width)
        {
            int mid = left + width - 1, right = left + 2*width - 1;
            if (mid >= n)
                continue;
            if (right >= n)
                right = n - 1;
            merge(a, tmp, left, mid, right);
        }
        if (width > n / 2)
            break;
    }
}

int main(void)
{
    int n;
    Item a[MAX];
    scanf("%d", &n);
    if (n < 0 || n > MAX)
        return 1;
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    mergeSort(a, n);
    printf("Sorted array: ");
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\nBest time complexity: O(n log n)\nWorst time complexity: O(n log n)\n");
    return 0;
}

9. Enter a integer array from the user and perform quick sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Code:

#include <stdio.h>
#define MAX 1000

typedef int Item;

int cmp(Item a, Item b)
{
    return (a > b) -(a < b);
}

void swap(Item *a, Item *b)
{
    int t = *a;
    *a = *b;
    *b = t;
}

int partition(Item a[], int low, int high)
{
    Item pivot;
    pivot = a[high];
    int i = low - 1;
    for (int j = low; j < high; j++)
        if (cmp(a[j], pivot) <= 0)
        {
            i++;
            swap(&a[i], &a[j]);
        }
    swap(&a[i + 1], &a[high]);
    return i + 1;
}

void quickSort(Item a[], int n)
{
    int stack[2*MAX], top = -1;
    if (n > 1)
    {
        stack[++top] = 0;
        stack[++top] = n - 1;
    }
    while (top >= 0)
    {
        int high = stack[top--], low = stack[top--];
        int p = partition(a, low, high);
        if (p - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = p - 1;
        }
        if (p + 1 < high)
        {
            stack[++top] = p + 1;
            stack[++top] = high;
        }
    }
}

int main(void)
{
    int n;
    Item a[MAX];
    scanf("%d", &n);
    if (n < 0 || n > MAX)
        return 1;
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    quickSort(a, n);
    printf("Sorted array: ");
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\nBest time complexity: O(n log n)\nWorst time complexity: O(n^2)\n");
    return 0;
}

void merge(Item a[], Item tmp[], int l, int m, int r)
{
    int i = l, j = m + 1, k = l;
    while (i <= m && j <= r)
    {
        if (cmp(a[i], a[j]) <= 0)
            strcpy(tmp[k++], a[i++]);
        else
            strcpy(tmp[k++], a[j++]);
    }
    while (i <= m)
        strcpy(tmp[k++], a[i++]);
    while (j <= r)
        strcpy(tmp[k++], a[j++]);
    for (int x = l; x <= r; x++)
        strcpy(a[x], tmp[x]);
}

{

int partition(Item a[], int low, int high)
{
    Item pivot;
    strcpy(pivot, a[high]);
    int i = low - 1;
    for (int j = low; j < high; j++)
        if (cmp(a[j], pivot) <= 0)
        {
            i++;
            swap(a[i], a[j]);
        }
    swap(a[i + 1], a[high]);
    return i + 1;
}

void quickSort(Item a[], int n)
{
    int stack[2*MAX], top = -1;
    if (n > 1)
    {
        stack[++top] = 0;
        stack[++top] = n - 1;
    }
    while (top >= 0)
    {
        int high = stack[top--], low = stack[top--];
        int p = partition(a, low, high);
        if (p - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = p - 1;
        }
        if (p + 1 < high)
        {
            stack[++top] = p + 1;
            stack[++top] = high;
        }
    }
}
