/* 1. Stack using an integer array: push, pop, peek, isEmpty, isFull */

int top = -1;

int isEmpty()
{
    return top == -1;
}

int isFull()
{
    return top == MAX - 1;
}

{
    if (isFull())
    {
        printf("Stack Overflow\n");
        return;

{
    if (isEmpty())
    {
        printf("Stack Underflow\n");
        return;
    }
    printf("%d popped\n", stack[top--]);
}

{
    if (isEmpty())
    {
        printf("Stack is empty\n");
        return;
    }
    printf("Top element: %d\n", stack[top]);
}

{
    int ch, x;
    do
    {
        printf("\n1.Push 2.Pop 3.Peek 4.IsEmpty 5.IsFull 6.Exit\nChoice: ");

        {
            case 1:
                printf("Enter value: ");

                printf(isFull() ? "Stack is full\n" : "Stack is not full\n");
                break;
            case 6:
                break;
            default:
                printf("Invalid choice\n");
        }
    } while (ch != 6);
    return 0;

3. Create a stack implementing the operations on an integer nodes (dynamically) (push, pop, peek, isempty, isfull)

/* 3. Stack using dynamically allocated integer nodes (linked list) */
#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *top = NULL;

int isEmpty()
{
    return top == NULL;
}

/* A linked stack is full only when memory cannot be allocated */

int isFull()
{
    struct node *t = (struct node *)malloc(sizeof(struct node));
    if (t == NULL)
        return 1;
    free(t);
    return 0;
}

void push(int x)
{
    if (isFull())
    {
        printf("Stack Overflow (no memory)\n");
        return;
    }
    struct node *n = (struct node *)malloc(sizeof(struct node));
    n->data = x;
    n->next = top;
    top = n;
    printf("%d pushed\n", x);
}

void pop()
{
    if (isEmpty())
    {
        printf("Stack Underflow\n");
        return;
    }
    struct node *t = top;
    printf("%d popped\n", t->data);
    top = top->next;
    free(t);
}

void peek()
{
    if (isEmpty())
    {
        printf("Stack is empty\n");
        return;
    }
    printf("Top element: %d\n", top->data);
}

{

                printf(isFull() ? "Stack is full\n" : "Stack is not full\n");
                break;
            case 6:
                break;

                printf("Invalid choice\n");
        }
    } while (ch != 6);
    while (!isEmpty())
    {
        struct node *t = top;
        top = top->next;
        free(t);
    }
    return 0;
}

5. Enter the string from the user and reverse the string using stacks.

Code:

/* 5. Reverse a string using a stack */
#include <stdio.h>
#include <string.h>
#define MAX 100

char stack[MAX];
int top = -1;

void push(char c)
{
    if (top < MAX - 1)
        stack[++top] = c;
}

char pop()
{
    return (top == -1) ? '\0' : stack[top--];
}

int main()
{
    char str[MAX];
    int i, n;
    printf("Enter a string: ");
    fgets(str, MAX, stdin);
    str[strcspn(str, "\n")] = '\0';
    n = strlen(str);
    for (i = 0; i < n; i++)
        push(str[i]);
    for (i = 0; i < n; i++)
        str[i] = pop();
    printf("Reversed string: %s\n", str);
    return 0;
}

6. Create the program to match the single bracket using stacks.

Code:

/* 6. Bracket matching using a stack: handles ( ) { } [ ] */
#include <stdio.h>
#define MAX 100

char stack[MAX];
int top = -1;

void push(char c)
{
    stack[++top] = c;
}

char pop()
{
    return stack[top--];
}

int isMatch(char open, char close)
{
    return (open == '(' && close == ')') || (open == '{' && close == '}') || (open == '[' && close == ']');
}

int main()
{
    char exp[MAX];
    int i, balanced = 1;
    printf("Enter an expression: ");
    scanf("%99s", exp);
    for (i = 0; exp[i] != '\0'; i++)
    {
        char c = exp[i];
        if (c == '(' || c == '{' || c == '[')
        {
            push(c);
        }
        else if (c == ')' || c == '}' || c == ']')
        {
            if (top == -1 || !isMatch(pop(), c))
            {
                balanced = 0;
                break;
            }
        }
    }
    if (top != -1)
        balanced = 0;
    printf(balanced ? "Brackets are balanced\n" : "Brackets are NOT balanced\n");
    return 0;
}

/* 7. Two integer stacks in a single array.
   Stack 1 grows from the left end, Stack 2 grows from the right end. */
#include <stdio.h>
#define MAX 10

int arr[MAX];
int top1 = -1;
int top2 = MAX;

void push(int s, int x)
{
    if (top1 + 1 == top2)
    {
        printf("Overflow: array is full\n");
        return;
    }
    if (s == 1)
        arr[++top1] = x;
    else
        arr[--top2] = x;
    printf("%d pushed to stack %d\n", x, s);
}

void pop(int s)
{
    if (s == 1)
    {
        if (top1 == -1)
        {
            printf("Stack 1 Underflow\n");
            return;
        }
        printf("%d popped from stack 1\n", arr[top1--]);
    }
    else
    {
        if (top2 == MAX)
        {
            printf("Stack 2 Underflow\n");
            return;
        }
        printf("%d popped from stack 2\n", arr[top2++]);
    }
}

void display()
{
    int i;
    printf("Stack 1 (top to bottom): ");
    for (i = top1; i >= 0; i--)
        printf("%d ", arr[i]);
    printf("\nStack 2 (top to bottom): ");
    for (i = top2; i < MAX; i++)
        printf("%d ", arr[i]);
    printf("\n");
}

int main()
{
    int ch, s, x;
    do
    {
        printf("\n1.Push 2.Pop 3.Display 4.Exit\nChoice: ");
        scanf("%d", &ch);
        switch (ch)
        {
            case 1:
                printf("Stack number (1/2): ");
                scanf("%d", &s);
                if (s != 1 && s != 2)
                {
                    printf("Invalid stack\n");
                    break;
                }
                printf("Enter value: ");
                scanf("%d", &x);
                push(s, x);
                break;
            case 2:
                printf("Stack number (1/2): ");
                scanf("%d", &s);
                if (s != 1 && s != 2)
                {
                    printf("Invalid stack\n");
                    break;
                }
                pop(s);
                break;
            case 3:
                display();
                break;
            case 4:
                break;
            default:
                printf("Invalid choice\n");
        }
    } while (ch != 4);
    return 0;
}

8. Sort the stack in ascending order using integer array.

Code:

/* 8. Sort a stack in ascending order using integer arrays (two stacks) */
#include <stdio.h>
#define MAX 50

int s[MAX], t[MAX]; /* s = input stack, t = temporary/result stack */
int ts = -1, tt = -1;

int main()
{
    int n, i, x;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    if (n > MAX)
        n = MAX;
    printf("Enter %d elements (last one becomes the top):\n", n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        s[++ts] = x;
    }
    while (ts != -1)
    {
        int tmp = s[ts--];
        while (tt != -1 && t[tt] > tmp)
            s[++ts] = t[tt--];
        t[++tt] = tmp;
    }
    printf("Sorted stack (bottom to top): ");
    for (i = 0; i <= tt; i++)
        printf("%d ", t[i]);
    printf("\n(smallest at bottom, largest at top)\n");
    return 0;
}

9. Enter the infix expression from user and convert into prefix expression.

Code:

/* 9. Infix to Prefix conversion using a stack */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 100

char stack[MAX];
int top = -1;

void push(char c)
{
    stack[++top] = c;
}

char pop()
{
    return stack[top--];
}

char peek()
{
    return stack[top];
}

int prec(char op)
{
    switch (op)
    {
        case '^':
            return 3;
        case '*':
        case '/':
        case '%':
            return 2;
        case '+':
        case '-':
            return 1;
    }
    return 0;
}

void reverse(char *s)
{
    int i, j;
    char t;
    for (i = 0, j = strlen(s) - 1; i < j; i++, j--)
    {
        t = s[i];
        s[i] = s[j];
        s[j] = t;
    }
}

int main()
{
    char infix[MAX], rev[MAX], out[MAX];
    int i, k = 0;
    printf("Enter infix expression: ");
    scanf("%99s", infix);
    strcpy(rev, infix);
    reverse(rev);
    for (i = 0; rev[i]; i++)
    {
        if (rev[i] == '(')
            rev[i] = ')';
        else if (rev[i] == ')')
            rev[i] = '(';
    }
    for (i = 0; rev[i]; i++)
    {
        char c = rev[i];
        if (isalnum((unsigned char)c))
        {
            out[k++] = c;
        }
        else if (c == '(')
        {
            push(c);
        }
        else if (c == ')')
        {
            while (top != -1 && peek() != '(')
                out[k++] = pop();
            if (top != -1)
                pop();
        }
        else
        {
            /* operator (associativity is flipped because the string is reversed) */
            if (c == '^')
                while (top != -1 && prec(peek()) >= prec(c))
                    out[k++] = pop();
            else
                while (top != -1 && prec(peek()) > prec(c))
                    out[k++] = pop();
            push(c);
        }
    }
    while (top != -1)
        out[k++] = pop();
    out[k] = '\0';
    reverse(out);
    printf("Prefix expression: %s\n", out);
    return 0;
}

Prefix expression: *+AB-CD

/* 11. Evaluate a prefix expression (single-digit operands), e.g. -*23+45 */

/* 12. Evaluate a postfix expression (single-digit operands), e.g. 23*45+- */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 100

int stack[MAX];
int top = -1;

void push(int x)
{
    stack[++top] = x;
}

int pop()
{
    return stack[top--];
}

int power(int a, int b)
{
    int r = 1;
    while (b-- > 0)
        r *= a;
    return r;
}

13. Implement the tower of hanoi using stacks.

Code:

/* 13. Tower of Hanoi using stacks (iterative, one stack per peg) */
#include <stdio.h>
#define MAX 20

int peg[3][MAX];
int top[3] = {-1, -1, -1 };
char name[3] = {'A', 'B', 'C' };

void push(int p, int d)
{
    peg[p][++top[p]] = d;
}

int pop(int p)
{
    return peg[p][top[p]--];
}

/* Make the only legal move between pegs a and b */

void move(int a, int b)
{
    int d;
    if (top[a] == -1)
    {
        d = pop(b);
        push(a, d);
        printf("Move disk %d from %c to %c\n", d, name[b], name[a]);
    }
    else if (top[b] == -1)
    {
        d = pop(a);
        push(b, d);
        printf("Move disk %d from %c to %c\n", d, name[a], name[b]);
    }
    else if (peg[a][top[a]] < peg[b][top[b]])
    {
        d = pop(a);
        push(b, d);
        printf("Move disk %d from %c to %c\n", d, name[a], name[b]);
    }
    else
    {
        d = pop(b);
        push(a, d);
        printf("Move disk %d from %c to %c\n", d, name[b], name[a]);
    }
}

int main()
{
    int n, i, src = 0, aux = 1, dest = 2, t;
    long total;
    printf("Enter number of disks (1-%d): ", MAX);
    scanf("%d", &n);
    if (n < 1 || n > MAX)
    {
        printf("Invalid number of disks\n");
        return 1;
    }
    for (i = n; i >= 1; i--)
        push(src, i);
    if (n % 2 == 0)
    {
        t = aux;
        aux = dest;
        dest = t;
    } /* even n: swap aux and dest */
    total = (1L << n) - 1;
    for (i = 1; i <= total; i++)
    {
        if (i % 3 == 1)
            move(src, dest);
        else if (i % 3 == 2)
            move(src, aux);
        else
            move(aux, dest);
    }
    printf("Total moves: %ld\n", total);
    return 0;
}
