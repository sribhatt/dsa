9. Enter an array from user and find the k-th smallest value?

Code:

#include <stdio.h>

int main()
{
    int a[100], n, i, j, k, temp;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("Enter k: ");
    scanf("%d", &k);
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - 1 - i; j++)
            if (a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
    if (k >= 1 && k <= n)
        printf("k = %d, k-th smallest value = %d\n", k, a[k - 1]);
    else
        printf("Invalid k\n");
    return 0;
}

Output:

Enter size of array: 6
Enter 6 elements: 45 12 89 3 67 21
Enter k: 3
k = 3, k-th smallest value = 21

14. Enter an array from user and print all possible combinations of the elements?

Code:

#include <stdio.h>

int main()
{
    int a[100], n, i, j, mask;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    if (n > 15)
    {
        printf("Use n <= 15\n");
        return 0;
    }
    printf("All combinations:\n");
    for (mask = 0; mask < (1 << n); mask++)
    {
        printf("{ ");
        for (j = 0; j < n; j++)
            if (mask & (1 << j))
                printf("%d ", a[j]);
        printf("}\n");
    }
    return 0;
}

15. Enter an array from user and print all the palindrome numbers?

Code:

#include <stdio.h>

int main()
{
    int a[100], n, i, num, rev, rem, found = 0;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("Palindrome numbers: ");
    for (i = 0; i < n; i++)
    {
        if (a[i] < 0)
            continue;
        num = a[i];
        rev = 0;
        while (num > 0)
        {
            rem = num % 10;
            rev = rev * 10 + rem;
            num = num / 10;
        }
        if (rev == a[i])
        {
            printf("%d ", a[i]);
            found = 1;
        }
    }
    if (found == 0)
        printf("None");
    printf("\n");
    return 0;
}

Output:

Enter size of array: 6
Enter 6 elements: 121 45 363 10 7 1331
Palindrome numbers: 121 363 7 1331

17. Merge the two sorted arrays into one array but in sorted order?

Code:

#include <stdio.h>

int main()
{
    int a[100], b[100], c[200], n1, n2, i, j = 0, k = 0;
    printf("Enter size of first sorted array: ");
    scanf("%d", &n1);
    printf("Enter %d elements: ", n1);
    for (i = 0; i < n1; i++)
        scanf("%d", &a[i]);
    printf("Enter size of second sorted array: ");
    scanf("%d", &n2);
    printf("Enter %d elements: ", n2);
    for (i = 0; i < n2; i++)
        scanf("%d", &b[i]);
    i = 0;
    while (i < n1 && j < n2)
    {
        if (a[i] <= b[j])
            c[k++] = a[i++];
        else
            c[k++] = b[j++];
    }
    while (i < n1)
        c[k++] = a[i++];
    while (j < n2)
        c[k++] = b[j++];
    printf("Merged array: ");
    for (i = 0; i < k; i++)
        printf("%d ", c[i]);
    printf("\n");
    return 0;
}

Output:

Enter 4 elements: 1 3 5 7
Enter size of second sorted array: 4
Enter 4 elements: 2 4 6 8
Merged array: 1 2 3 4 5 6 7 8

18. Find the local minima in the array? (An element in an array is a local minima if it less than the element after it, and the element before it)

Code:

#include <stdio.h>

int main()
{
    int a[100], n, i, found = 0;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("Local minima: ");
    for (i = 1; i < n - 1; i++)
        if (a[i] < a[i - 1] && a[i] < a[i + 1])
        {
            printf("%d ", a[i]);
            found = 1;
        }
    if (found == 0)
        printf("None");
    printf("\n");
    return 0;
}

20. Trapping Rain Water Problem: Given n non-negative integers representing an elevation map where the width of each bar is 1, compute how much water it can trap after raining.

Code:

#include <stdio.h>

int main()
{
    int a[100], n, i, left[100], right[100], minh, water = 0;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    if (n < 3)
    {
        printf("Water trapped = 0\n");
        return 0;
    }
    left[0] = a[0];
    for (i = 1; i < n; i++)
        left[i] = (a[i] > left[i - 1]) ? a[i] : left[i - 1];
    right[n - 1] = a[n - 1];
    for (i = n - 2; i >= 0; i--)
        right[i] = (a[i] > right[i + 1]) ? a[i] : right[i + 1];
    for (i = 0; i < n; i++)
    {
        minh = (left[i] < right[i]) ? left[i] : right[i];
        water = water + (minh - a[i]);
    }
    printf("Water trapped = %d units\n", water);
    return 0;
}

Output:

Enter size of array: 12
Enter 12 elements: 0 1 0 2 1 0 1 3 2 1 2 1
Water trapped = 6 units
