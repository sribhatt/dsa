1. Calculate factorial of a number using recursion

Code:

#include <stdio.h>

long long fac(int n)
{
    if (n <= 1)
        return 1;
    return n * fac(n - 1);
}

int main()
{
    int n;
    printf("ENTER a number: ");
    scanf("%d", &n);
    if (n < 0)
        printf("factorial Not defined for negative numbers\n");
    else
        printf("Factorial of %d is = %lld\n", n, fac(n));
    return 0;
}

Output:

ENTER a number: 5
Factorial of 5 is = 120

2. Calculate fibonacci of a number using recursion

Code:

#include <stdio.h>

int fibonic(int n)
{
    if (n == 0)
        return 0;
    if (n == 1)
        return 1;
    return fibonic(n - 1) + fibonic(n - 2);
}

int main()
{
    int n;
    printf("enter The number: ");
    scanf("%d", &n);
    printf("FIBONACCI of %d IS %d\n", n, fibonic(n));
    return 0;
}

Output:

enter The number: 7
FIBONACCI of 7 IS 13

3. Calculate the sum of natural numbers till 10 using recursion.

Code:

#include <stdio.h>

int sum(int n)
{
    if (n == 0)
        return 0;
    return n + sum(n - 1);
}

int main()
{
    printf("sum of Natural numbers till 10 IS: %d\n", sum(10));
    return 0;
}

Output:

sum of Natural numbers till 10 IS: 55

4. Convert a number into binary using recursion

#include <stdio.h>

{

5. Find the largest element in the integer array using recursion.

Code:

#include <stdio.h>

int largest(int a[], int n)
{
    int m;
    if (n == 1)
        return a[0];
    m = largest(a, n - 1);
    return (a[n - 1] > m) ? a[n - 1] : m;
}

int main()
{
    int n, i, a[100];
    printf("how many ELEMENTS: ");
    scanf("%d", &n);
    printf("enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("LARGEST element Is: %d\n", largest(a, n));
    return 0;
}

Output:

how many ELEMENTS: 5
enter elements: 12 45 7 89 23
LARGEST element Is: 89

7. Enter the number from the user and add all digits of the number using recursion.

Code:

#include <stdio.h>

int digitSum(int n)
{
    if (n == 0)
        return 0;
    return (n % 10) + digitSum(n / 10);
}

int main()
{
    int n;
    printf("Enter the NUMBER: ");
    scanf("%d", &n);
    if (n < 0)
        n = -n;
    printf("SUM of DIGITS is = %d\n", digitSum(n));
    return 0;
}

Enter the NUMBER: 12345
SUM of DIGITS is = 15

8. Find the arithmetic mean of an integer array using recursion.

Code:

#include <stdio.h>

int sumArray(int a[], int n)
{
    if (n == 0)
        return 0;
    return a[n - 1] + sumArray(a, n - 1);
}

int main()
{
    int n, i, a[100];
    printf("number OF elements: ");
    scanf("%d", &n);
    printf("Enter ELEMENTS: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("arithmetic MEAN is: %.2f\n", (float)sumArray(a, n) / n);
    return 0;
}

Output:

number OF elements: 5
Enter ELEMENTS: 10 20 30 40 50
arithmetic MEAN is: 30.00
