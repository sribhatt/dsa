1. Construct binary tree from preorder and inorder traversal

Code:

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *newNode(int x)
{
    struct Node *n = (struct Node *)malloc(sizeof(struct Node));
    n->data = x;
    n->left = NULL;
    n->right = NULL;
    return n;
}

/* find position of x in inorder array */

int findIndex(int in[], int start, int end, int x)
{
    int i;
    for (i = start; i <= end; i++)
    {
        if (in[i] == x)
        {
            return i;
        }
    }
    return -1;
}

/* build tree from preorder and inorder */

struct Node *build(int pre[], int in[], int *pi, int start, int end)
{
    int x, k;
    struct Node *root;
    if (start > end)
    {
        return NULL;
    }
    x = pre[*pi];
    (*pi)++;
    root = newNode(x);
    k = findIndex(in, start, end, x);
    if (k == -1)
    {
        return root;
    }
    root->left = build(pre, in, pi, start, k - 1);
    root->right = build(pre, in, pi, k + 1, end);
    return root;
}

void levelOrder(struct Node *root)
{
    struct Node *q[10000];
    struct Node *cur;
    int front = 0, rear = 0;
    if (root == NULL)
    {
        return;
    }
    q[rear++] = root;
    while (front < rear)
    {
        cur = q[front++];
        printf("%d ", cur->data);
        if (cur->left != NULL)
        {
            q[rear++] = cur->left;
        }
        if (cur->right != NULL)
        {
            q[rear++] = cur->right;
        }
    }
}

int main()
{
    int n, i, pi = 0;
    int pre[1000], in[1000];
    struct Node *root;
    scanf("%d", &n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &pre[i]);
    }
    for (i = 0; i < n; i++)
    {
        scanf("%d", &in[i]);
    }
    root = build(pre, in, &pi, 0, n - 1);
    levelOrder(root);
    return 0;
}

int main()
{
    int n, x, i, l, r;
    int front = 0, rear = 0;
    struct Node **nodes;
    struct Node *root, *cur;
    struct Node *q[10000];
    scanf("%d", &n);
    if (n <= 0)
    {
        return 0;
    }
    nodes = (struct Node **)malloc(n * sizeof(struct Node *));
    /* -1 means NULL node */
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        if (x == -1)
        {
            nodes[i] = NULL;
        }
        else
        {
            nodes[i] = newNode(x);
        }
    }
    /* connect children: left = 2i+1, right = 2i+2 */
    for (i = 0; i < n; i++)
    {
        if (nodes[i] != NULL)
        {
            l = 2 * i + 1;
            r = 2 * i + 2;
            if (l < n)
            {
                nodes[i]->left = nodes[l];
            }
            if (r < n)
            {
                nodes[i]->right = nodes[r];
            }
        }
    }
    root = nodes[0];
    /* level order traversal */
    if (root != NULL)
    {
        q[rear++] = root;
    }
    while (front < rear)
    {
        cur = q[front++];
        printf("%d ", cur->data);
        if (cur->left != NULL)
        {
            q[rear++] = cur->left;
        }
        if (cur->right != NULL)
        {
            q[rear++] = cur->right;
        }

    return 0;
}

int depth(struct Node *root)
{
    int l, r;
    if (root == NULL)
    {
        return 0;
    }
    l = depth(root->left);
    r = depth(root->right);

    {
        return 1 + l;
    }
    else
    {
        return 1 + r;
    }
}

7. Create binary search tree using arrays (insert, delete, search, preorder, inorder, postorder, minimum element)

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *newNode(int x)
{
    struct Node *n = (struct Node *)malloc(sizeof(struct Node));
    n->data = x;
    n->left = NULL;
    n->right = NULL;
    return n;
}

struct Node *insert(struct Node *root, int x)
{
    if (root == NULL)
    {
        return newNode(x);
    }
    if (x < root->data)
    {
        root->left = insert(root->left, x);
    }
    else if (x > root->data)
    {
        root->right = insert(root->right, x);
    }
    return root;
}

struct Node *minimum(struct Node *root)
{
    while (root != NULL && root->left != NULL)
    {
        root = root->left;
    }
    return root;
}

struct Node *deleteNode(struct Node *root, int x)
{
    struct Node *t;
    if (root == NULL)
    {
        return NULL;
    }
    if (x < root->data)
    {
        root->left = deleteNode(root->left, x);
    }
    else if (x > root->data)
    {
        root->right = deleteNode(root->right, x);
    }
    else
    {
        /* node with 0 or 1 child */
        if (root->left == NULL)
        {
            t = root->right;
            free(root);
            return t;
        }
        if (root->right == NULL)
        {
            t = root->left;
            free(root);
            return t;
        }
        /* node with 2 children: use inorder successor */
        t = minimum(root->right);
        root->data = t->data;
        root->right = deleteNode(root->right, t->data);
    }
    return root;
}

int search(struct Node *root, int x)
{
    while (root != NULL)
    {
        if (root->data == x)
        {
            return 1;
        }
        if (x < root->data)
        {
            root = root->left;
        }
        else
        {
            root = root->right;
        }
    }
    return 0;
}

void preorder(struct Node *r)
{
    if (r != NULL)
    {
        printf("%d ", r->data);
        preorder(r->left);
        preorder(r->right);
    }
}

void inorder(struct Node *r)
{
    if (r != NULL)
    {
        inorder(r->left);
        printf("%d ", r->data);
        inorder(r->right);
    }
}

void postorder(struct Node *r)
{
    if (r != NULL)
    {
        postorder(r->left);
        postorder(r->right);
        printf("%d ", r->data);
    }
}

9. Find inorder predecessor and inorder successor of binary search tree (dynamically represented)

Code:

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *insert(struct Node *r, int x)
{
    struct Node *n;
    if (r == NULL)
    {
        n = (struct Node *)malloc(sizeof(struct Node));
        n->data = x;
        n->left = NULL;
        n->right = NULL;
        return n;
    }
    if (x < r->data)
    {
        r->left = insert(r->left, x);
    }
    else if (x > r->data)
    {
        r->right = insert(r->right, x);
    }
    return r;
}

int main()
{
    struct Node *root = NULL;
    struct Node *cur, *pred = NULL, *succ = NULL, *t;
    int n, x, i, key;
    scanf("%d", &n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        root = insert(root, x);
    }
    scanf("%d", &key);
    cur = root;
    while (cur != NULL)
    {
        if (key < cur->data)
        {
            succ = cur;
            cur = cur->left;
        }
        else if (key > cur->data)
        {
            pred = cur;
            cur = cur->right;
        }
        else
        {
            /* key found: predecessor = rightmost of left subtree */
            if (cur->left != NULL)
            {
                t = cur->left;
                while (t->right != NULL)
                {
                    t = t->right;
                }
                pred = t;
            }
            /* successor = leftmost of right subtree */
            if (cur->right != NULL)
            {
                t = cur->right;
                while (t->left != NULL)
                {
                    t = t->left;
                }
                succ = t;
            }
            break;
        }
    }
    if (pred != NULL)
    {
        printf("Predecessor: %d\n", pred->data);
    }
    else
    {
        printf("Predecessor: NONE\n");
    }
    if (succ != NULL)
    {
        printf("Successor: %d\n", succ->data);
    }
    else
    {
        printf("Successor: NONE\n");
    }
    return 0;
}

10. Check whether binary search tree contains dead end

Code:

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *insert(struct Node *r, int x)
{
    struct Node *n;
    if (r == NULL)
    {
        n = (struct Node *)malloc(sizeof(struct Node));
        n->data = x;
        n->left = NULL;
        n->right = NULL;
        return n;
    }
    if (x < r->data)
    {
        r->left = insert(r->left, x);
    }
    else if (x > r->data)
    {
        r->right = insert(r->right, x);
    }
    return r;
}

/* dead end: no more value can be inserted below a node (low == high) */

int deadEnd(struct Node *r, long long low, long long high)
{
    if (r == NULL)
    {
        return 0;
    }
    if (low == high)
    {
        return 1;
    }
    return deadEnd(r->left, low, r->data - 1) || deadEnd(r->right, r->data + 1, high);
}

int main()
{
    struct Node *root = NULL;
    int n, x, i;
    scanf("%d", &n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        root = insert(root, x);
    }
    if (deadEnd(root, 1, 2147483647LL))
    {
        printf("Dead End Found");
    }
    else
    {
        printf("No Dead End");
    }
    return 0;
}

struct Node *newNode(int x)
{

    n->data = x;
    n->left = NULL;
    n->right = NULL;
    return n;
}

/* compare function for qsort */

int cmp(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;
    if (x > y)
    {
        return 1;
    }
    else if (x < y)
    {
        return -1;
    }
    return 0;
}

/* store inorder values in array */

void inorderStore(struct Node *r, int a[], int *i)
{
    if (r != NULL)
    {
        inorderStore(r->left, a, i);
        a[*i] = r->data;
        (*i)++;
        inorderStore(r->right, a, i);
    }
}

/* put sorted values back in inorder */

void inorderFill(struct Node *r, int a[], int *i)
{
    if (r != NULL)
    {
        inorderFill(r->left, a, i);
        r->data = a[*i];
        (*i)++;
        inorderFill(r->right, a, i);
    }
}

{

    {
        printf("%d ", r->data);
        preorder(r->left);
        preorder(r->right);
    }
}

    k = 0;

    {
        return NULL;
    }
    n = (struct Node *)malloc(sizeof(struct Node));
    n->data = x;
    printf("Enter left child of %d (-1 for NULL): ", x);
    n->left = create();
    printf("Enter right child of %d (-1 for NULL): ", x);
    n->right = create();
    return n;
}

/* strict binary tree: every node has 0 or 2 children */

int strict(struct Node *r)
{
    if (r == NULL)
    {
        return 1;
    }
    if ((r->left == NULL && r->right != NULL) || (r->left != NULL && r->right == NULL))
    {
        return 0;
    }
    return strict(r->left) && strict(r->right);
}

void preorder(struct Node *r)
{
    if (r != NULL)
    {
        printf("%d ", r->data);
        preorder(r->left);
        preorder(r->right);
    }
}

    printf("Enter root value (-1 for NULL): ");
    r = create();
    printf("Preorder: ");
    preorder(r);

    {
        printf("\nStrict binary tree");
    }
    else
    {
        printf("\nNot a strict binary tree");
    }
