1. Implement the operations of simple queue in an integer array (insert, delete, peek, isempty, isfull)

Code:

#include <stdio.h>
#include <stdlib.h>

enum
{
    MAX = 5
};

int q[MAX];
int front = 0, rear = -1; /* simple (linear) queue: freed slots are not reused */

int isEmpty()
{
    return front > rear;
}

int isFull()
{
    return rear == MAX - 1;
}

void insertItem(int val)
{
    int idx;
    if (isFull())
    {
        printf("Queue is full (overflow)\n");
        return;
    }
    idx = ++rear;
    q[idx] = val;
    printf("Inserted\n");
}

void deleteItem()
{
    if (isEmpty())
    {
        printf("Queue is empty (underflow)\n");
        return;
    }
    printf("Deleted: %d\n", q[front]);
    front++;
}

void peek()
{
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Front element: %d\n", q[front]);
}

void display()
{
    int i;
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Queue (front to rear): ");
    for (i = front; i <= rear; i++)
        printf("%d ", q[i]);
    printf("\n");
}

int main()
{
    int ch;
    int v;
    while (1)
    {
        printf("\n1.Insert\n2.Delete\n3.Peek\n4.IsEmpty\n5.IsFull\n6.Display\n7.Exit\nEnter choice: ");
        if (scanf("%d", &ch) != 1)
            break;
        switch (ch)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &v);
                insertItem(v);
                break;
            case 2:
                deleteItem();
                break;
            case 3:
                peek();
                break;
            case 4:
                printf(isEmpty() ? "Queue is empty\n" : "Queue is not empty\n");
                break;
            case 5:
                printf(isFull() ? "Queue is full\n" : "Queue is not full\n");
                break;
            case 6:
                display();
                break;
            case 7:
                return 0;
            default:
                printf("Invalid choice\n");
        }
    }
    return 0;
}

3. Implement the operations of queue dynamically of integers (insert, delete, peek, isempty, isfull)

Code:

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *front = NULL, *rear = NULL;

int isEmpty()
{
    return front == NULL;
}

int isFull()
{
    struct node *t = (struct node *)malloc(sizeof(struct node));
    if (t == NULL)
        return 1;
    free(t);
    return 0;
}

void insertItem(int val)
{
    struct node *n;
    if (isFull())
    {
        printf("Queue is full (overflow)\n");
        return;
    }
    n = (struct node *)malloc(sizeof(struct node));
    n->data = val;
    n->next = NULL;
    if (rear == NULL)
        front = rear = n;
    else
    {
        rear->next = n;
        rear = n;
    }
    printf("Inserted\n");
}

void deleteItem()
{
    struct node *t;
    if (isEmpty())
    {
        printf("Queue is empty (underflow)\n");
        return;
    }
    t = front;
    printf("Deleted: %d\n", t->data);
    front = front->next;
    if (front == NULL)
        rear = NULL;
    free(t);
}

void peek()
{
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Front element: %d\n", front->data);
}

void display()
{
    struct node *t = front;
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Queue (front to rear): ");
    while (t)
    {
        printf("%d ", t->data);
        t = t->next;
    }
    printf("\n");
}

int main()
{
    int ch;
    int v;
    while (1)
    {
        printf("\n1.Insert\n2.Delete\n3.Peek\n4.IsEmpty\n5.IsFull\n6.Display\n7.Exit\nEnter choice: ");
        if (scanf("%d", &ch) != 1)
            break;
        switch (ch)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &v);
                insertItem(v);
                break;
            case 2:
                deleteItem();
                break;
            case 3:
                peek();
                break;
            case 4:
                printf(isEmpty() ? "Queue is empty\n" : "Queue is not empty\n");
                break;
            case 5:
                printf(isFull() ? "Queue is full\n" : "Queue is not full\n");
                break;
            case 6:
                display();
                break;
            case 7:
                return 0;
            default:
                printf("Invalid choice\n");
        }
    }
    return 0;
}

5. Implement the operations of circular queue in an integer array (insert, delete, peek, isempty, isfull)

#include <stdlib.h>

{
    MAX = 5
};

int front = 0, count = 0;

{

        printf("Queue is full (overflow)\n");
        return;

    idx = (front + count) % MAX;
    q[idx] = val;
    count++;

{

        printf("Queue is empty\n");
        return;
    }
    printf("Queue (front to rear): ");

    printf("\n");
}

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    int pri;
    struct node *next;
};

struct node *front = NULL; /* sorted list: front has the highest priority (lowest number) */

int isEmpty()
{
    return front == NULL;
}

int isFull()
{
    struct node *t = (struct node *)malloc(sizeof(struct node));
    if (t == NULL)
        return 1;
    free(t);
    return 0;
}

void insertItem(int val, int pri)
{
    struct node *n, *t;
    if (isFull())
    {
        printf("Queue is full (overflow)\n");
        return;
    }
    n = (struct node *)malloc(sizeof(struct node));
    n->data = val;
    n->pri = pri;
    if (front == NULL || pri < front->pri)
    {
        n->next = front;
        front = n;
    }
    else
    {
        t = front;
        while (t->next && t->next->pri <= pri)
            t = t->next;
        n->next = t->next;
        t->next = n;
    }
    printf("Inserted\n");
}

void deleteItem()
{
    struct node *t;
    if (isEmpty())
    {
        printf("Queue is empty (underflow)\n");
        return;
    }
    t = front;
    printf("Deleted: %d (priority %d)\n", t->data, t->pri);
    front = front->next;
    free(t);
}

void peek()
{
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Highest priority: %d (priority %d)\n", front->data, front->pri);
}

void display()
{
    struct node *t = front;
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Queue: ");
    while (t)
    {
        printf("%d(%d) ", t->data, t->pri);
        t = t->next;
    }
    printf("\n");
}

12. Implement the operations of priority queue dynamically of string (insert, delete, peek, isempty, isfull)

Code:

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[50];
    int pri;
    struct node *next;
};

struct node *front = NULL; /* sorted list: front has the highest priority (lowest number) */

int isEmpty()
{
    return front == NULL;
}

int isFull()
{
    struct node *t = (struct node *)malloc(sizeof(struct node));
    if (t == NULL)
        return 1;
    free(t);
    return 0;
}

void insertItem(const char *val, int pri)
{
    struct node *n, *t;
    if (isFull())
    {
        printf("Queue is full (overflow)\n");
        return;
    }
    n = (struct node *)malloc(sizeof(struct node));
    strcpy(n->data, val);
    n->pri = pri;
    if (front == NULL || pri < front->pri)
    {
        n->next = front;
        front = n;
    }
    else
    {
        t = front;
        while (t->next && t->next->pri <= pri)
            t = t->next;
        n->next = t->next;
        t->next = n;
    }
    printf("Inserted\n");
}

void deleteItem()
{
    struct node *t;
    if (isEmpty())
    {
        printf("Queue is empty (underflow)\n");
        return;
    }
    t = front;
    printf("Deleted: %s (priority %d)\n", t->data, t->pri);
    front = front->next;
    free(t);
}

void peek()
{
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Highest priority: %s (priority %d)\n", front->data, front->pri);
}

void display()
{
    struct node *t = front;
    if (isEmpty())
    {
        printf("Queue is empty\n");
        return;
    }
    printf("Queue: ");
    while (t)
    {
        printf("%s(%d) ", t->data, t->pri);
        t = t->next;
    }
    printf("\n");
}

int main()
{
    int ch, pr;
    char v[50];
    printf("(Lower priority number = higher priority)\n");
    while (1)
    {
        printf("\n1.Insert\n2.Delete\n3.Peek\n4.IsEmpty\n5.IsFull\n6.Display\n7.Exit\nEnter choice: ");
        if (scanf("%d", &ch) != 1)
            break;
        switch (ch)
        {
            case 1:
                printf("Enter value: ");
                scanf("%49s", v);
                printf("Enter priority: ");
                scanf("%d", &pr);
                insertItem(v, pr);
                break;
            case 2:
                deleteItem();
                break;
            case 3:
                peek();
                break;
            case 4:
                printf(isEmpty() ? "Queue is empty\n" : "Queue is not empty\n");
                break;
            case 5:
                printf(isFull() ? "Queue is full\n" : "Queue is not full\n");
                break;
            case 6:
                display();
                break;
            case 7:
                return 0;
            default:
                printf("Invalid choice\n");
        }
    }
    return 0;
}

13. Find the first circular tour that visits all petrol pumps

Code:

#include <stdio.h>

/* Circular tour: find the first petrol pump from which a truck can
   complete the full circle. Prints -1 if no such pump exists. */

int main()
{
    int n, i, start = 0, tank = 0, total = 0;
    printf("Enter number of petrol pumps: ");
    scanf("%d", &n);
    int petrol[n], dist[n];
    for (i = 0; i < n; i++)
    {
        printf("Pump %d - petrol and distance to next pump: ", i);
        scanf("%d %d", &petrol[i], &dist[i]);
    }
    for (i = 0; i < n; i++)
    {
        int diff = petrol[i] - dist[i];
        total += diff;
        tank += diff;
        if (tank < 0)
        {
            start = i + 1;
            tank = 0;
        } /* restart from the next pump */
    }
    if (total >= 0)
        printf("First circular tour starts at pump index: %d\n", start);
    else
        printf("-1 (no circular tour possible)\n");
    return 0;
}

#include <stdio.h>
#include <string.h>

/* Length of the longest valid (well-formed) parentheses substring */

int main()
{
    char s[10001];
    int stack[10002], top = -1, i, n, max = 0;
    printf("Enter string of ( and ): ");
    scanf("%10000s", s);
    n = strlen(s);
    stack[++top] = -1; /* base index */
    for (i = 0; i < n; i++)
    {
        if (s[i] == '(')
            stack[++top] = i;
        else
        {
            top--;
            if (top == -1)
                stack[++top] = i; /* new base */
            else if (i - stack[top] > max)
                max = i - stack[top];
        }
    }
    printf("Longest valid substring length: %d\n", max);
    return 0;
}

#include <string.h>

int main()
{

    int cnt[256] = {0 }, front = 0, rear = 0, i, n;
    printf("Enter the stream (lowercase letters): ");
    scanf("%1000s", s);
    n = strlen(s);
    printf("Output: ");

    {
        cnt[(unsigned char)s[i]]++;
        q[rear++] = s[i]; /* enqueue */
        while (front < rear && cnt[(unsigned char)q[front]] > 1)
            front++; /* dequeue repeats */
        printf("%c ", front == rear ? '#' : q[front]);
    }

}

18. Reverse First K elements of Queue

Code:

#include <stdio.h>

enum
{
    MAX = 100
};

int q[MAX], front = 0, count = 0;
int st[MAX], top = -1;

void enqueue(int x)
{
    q[(front + count) % MAX] = x;
    count++;
}

int dequeue()
{
    int x = q[front];
    front = (front + 1) % MAX;
    count--;
    return x;
}

/* Reverse the first K elements of the queue */

int main()
{
    int n, k, i, x;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &x);
        enqueue(x);
    }
    printf("Enter K: ");
    scanf("%d", &k);
    if (k < 0 || k > n)
    {
        printf("Invalid K\n");
        return 0;
    }
    for (i = 0; i < k; i++)
        st[++top] = dequeue(); /* first K into stack */
    while (top != -1)
        enqueue(st[top--]); /* back in reversed order */
    for (i = 0; i < n - k; i++)
        enqueue(dequeue()); /* rotate remaining n-K */
    printf("Queue after reversing first %d elements: ", k);
    while (count > 0)
        printf("%d ", dequeue());
    printf("\n");
    return 0;
}

20. Rotten Oranges

#include <stdio.h>

#define MAX 10

int grid[MAX][MAX];
int qr[MAX * MAX], qc[MAX * MAX];

int main()
{
    int r, c, i, j, d, size, front = 0, rear = 0, fresh = 0, minutes = 0;
    int dr[4] = {-1, 1, 0, 0 };
    int dc[4] = {0, 0, -1, 1 };
    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);
    printf("Enter grid (0 = empty, 1 = fresh, 2 = rotten):\n");
    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            scanf("%d", &grid[i][j]);
            if (grid[i][j] == 2)
            {
                qr[rear] = i;
                qc[rear] = j;
                rear++;
            }
            else if (grid[i][j] == 1)
            {
                fresh++;
            }
        }
    }
    while (front < rear && fresh > 0)
    {
        size = rear - front;
        while (size--)
        {
            int x = qr[front], y = qc[front];
            front++;
            for (d = 0; d < 4; d++)
            {
                int nx = x + dr[d], ny = y + dc[d];
                if (nx >= 0 && nx < r && ny >= 0 && ny < c && grid[nx][ny] == 1)
                {
                    grid[nx][ny] = 2;
                    fresh--;
                    qr[rear] = nx;
                    qc[rear] = ny;
                    rear++;
                }
            }
        }
        minutes++;
    }
    if (fresh == 0)
        printf("Minimum time to rot all oranges: %d minutes\n", minutes);
    else
        printf("-1 (some oranges can never rot)\n");
    return 0;
}
